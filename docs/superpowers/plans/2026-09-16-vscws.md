# vscws Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A small bash tool `vscws` that generates per-project VS Code dev-container workspaces from language presets, plus the presets and a bullet-point README.

**Architecture:** One bash script (`bin/vscws`) merges `presets/_common.json` with a preset's `devcontainer.json` using `jq` and writes the result into `$VSCWS_ROOT/NAME/.devcontainer/`. Presets are plain JSON (and optional Dockerfiles). VS Code builds and runs the containers; the tool never builds anything.

**Tech Stack:** bash 3.2+, jq, git, Dev Container spec (devcontainer.json, features), Docker.

**Spec:** `docs/superpowers/specs/2026-09-16-vscws-design.md`

## Global Constraints

- `bin/vscws` must run under bash 3.2 (macOS default): no associative arrays, no `mapfile`, no `${var,,}`, no `readlink -f`, no `realpath`.
- Runtime dependencies of the tool: bash, jq, git (only for `--repo`), docker (only for `rm`). Nothing else.
- The tool never installs software, never edits shell profiles, never builds images.
- Nothing version-pinned in presets: every tool resolves to latest stable at image build time.
- Every error is one line on stderr prefixed `vscws: ` with exit code 1.
- Generation happens in a temp dir inside `$VSCWS_ROOT` and is moved into place with `mv`.
- Commit after every task with a `git -c user.name="Marat Satskevich" -c user.email="taram.msatskevich@gmail.com" commit` and the trailer `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- Repo root: `/home/msatskevich/vscws`. Run tests with `bash tests/run.sh` from the repo root.

`bin/vscws`:

```bash
#!/usr/bin/env bash
# vscws — create per-project VS Code dev container workspaces from presets.
# Compatible with bash 3.2 (macOS default). Dependencies: jq; git for --repo; docker for rm.
set -euo pipefail

VSCWS_VERSION="0.1.0"

die()  { printf 'vscws: %s\n' "$*" >&2; exit 1; }
info() { printf '%s\n' "$*"; }

usage() {
  cat <<'EOF'
Usage: vscws <command> [args]

Commands:
  new NAME --preset P [--repo URL]   create workspace NAME from preset P
  ls                                 list workspaces
  presets                            list available presets
  open NAME                          print (or run) the VS Code command to open NAME
  rm NAME                            delete workspace NAME (asks you to type the name)
  --version, -v                      print version
  --help, -h                         this help

Config file: ~/.config/vscws/config with VSCWS_ROOT, VSCWS_SSH_HOST,
VSCWS_CLAUDE_DIR (default ~/.claude), VSCWS_MODE (remote|local, auto-detected).
EOF
}

# Directory of the real script, following symlinks without readlink -f (macOS).
resolve_self_dir() {
  local src="$0" dir
  while [ -L "$src" ]; do
    dir="$(cd "$(dirname "$src")" && pwd -P)"
    src="$(readlink "$src")"
    case "$src" in /*) ;; *) src="$dir/$src" ;; esac
  done
  cd "$(dirname "$src")" && pwd -P
}

BIN_DIR="$(resolve_self_dir)"
REPO_DIR="$(cd "$BIN_DIR/.." && pwd -P)"
PRESETS_DIR="${VSCWS_PRESETS_DIR:-$REPO_DIR/presets}"
CONFIG_FILE="${VSCWS_CONFIG:-$HOME/.config/vscws/config}"

need() { command -v "$1" >/dev/null 2>&1 || die "'$1' is required but not installed (see README: Prerequisites)"; }

# Single-quote a string for writing into the shell-sourceable config file.
sq() { printf "'%s'" "$(printf '%s' "$1" | sed "s/'/'\\\\''/g")"; }

default_mode() { case "$(uname -s)" in Darwin) echo local ;; *) echo remote ;; esac; }

# Load ~/.config/vscws/config. Environment variables win over the file.
load_config() {
  local root="${VSCWS_ROOT:-}" host="${VSCWS_SSH_HOST:-}" cdir="${VSCWS_CLAUDE_DIR:-}" mode="${VSCWS_MODE:-}"
  if [ -f "$CONFIG_FILE" ]; then
    # shellcheck disable=SC1090
    . "$CONFIG_FILE"
  fi
  if [ -n "$root" ]; then VSCWS_ROOT="$root"; fi
  if [ -n "$host" ]; then VSCWS_SSH_HOST="$host"; fi
  if [ -n "$cdir" ]; then VSCWS_CLAUDE_DIR="$cdir"; fi
  if [ -n "$mode" ]; then VSCWS_MODE="$mode"; fi
  VSCWS_ROOT="${VSCWS_ROOT:-}"
  VSCWS_SSH_HOST="${VSCWS_SSH_HOST:-}"
  VSCWS_CLAUDE_DIR="${VSCWS_CLAUDE_DIR:-$HOME/.claude}"
  VSCWS_MODE="${VSCWS_MODE:-$(default_mode)}"
  case "$VSCWS_MODE" in remote|local) ;; *) die "VSCWS_MODE must be 'remote' or 'local' (got '$VSCWS_MODE')" ;; esac
}

# Make sure VSCWS_ROOT is known. On the first interactive run, ask and write the config file.
ensure_root() {
  if [ -n "$VSCWS_ROOT" ]; then return 0; fi
  if [ ! -t 0 ]; then die "VSCWS_ROOT is not set; create $CONFIG_FILE (see README: Install vscws)"; fi
  local root host=""
  printf 'Workspace root directory [%s/ws]: ' "$HOME"
  read -r root
  root="${root:-$HOME/ws}"
  case "$root" in "~/"*) root="$HOME/${root#\~/}" ;; "~") root="$HOME" ;; esac
  if [ "$VSCWS_MODE" = remote ]; then
    printf 'SSH host alias your laptop uses for this machine (for "vscws open", may be empty): '
    read -r host
  fi
  mkdir -p "$(dirname "$CONFIG_FILE")"
  {
    printf 'VSCWS_ROOT=%s\n' "$(sq "$root")"
    printf 'VSCWS_SSH_HOST=%s\n' "$(sq "$host")"
    printf 'VSCWS_CLAUDE_DIR=%s\n' "$(sq "$VSCWS_CLAUDE_DIR")"
  } > "$CONFIG_FILE"
  info "wrote $CONFIG_FILE"
  VSCWS_ROOT="$root"
  VSCWS_SSH_HOST="$host"
}

ws_path() { printf '%s/%s' "$VSCWS_ROOT" "$1"; }

validate_name() {
  case "$1" in ""|.*|-*) die "invalid workspace name '$1'" ;; esac
  printf '%s' "$1" | grep -Eq '^[A-Za-z0-9._-]+$' || die "invalid workspace name '$1' (allowed: letters, digits, . _ -)"
}

preset_dir() { printf '%s/%s' "$PRESETS_DIR" "$1"; }

main() {
  if [ $# -eq 0 ]; then usage >&2; exit 1; fi
  local cmd="$1"; shift
  case "$cmd" in
    -h|--help|help)       usage ;;
    -v|--version|version) info "vscws $VSCWS_VERSION" ;;
    presets)              cmd_presets "$@" ;;
    new|ls|open|rm)       load_config; "cmd_$cmd" "$@" ;;
    *)                    die "unknown command '$cmd' (try: vscws --help)" ;;
  esac
}

main "$@"
```

```bash
# generate_devcontainer PRESET NAME OUTDIR
# Deep-merges presets/_common.json with the preset, then applies name, Claude dir and mode.
generate_devcontainer() {
  local preset="$1" name="$2" out="$3" pdir f
  pdir="$(preset_dir "$preset")"
  mkdir -p "$out"
  jq -s \
    --arg name "$name" \
    --arg claude_dir "$VSCWS_CLAUDE_DIR" \
    --arg mode "$VSCWS_MODE" '
    def dedupe: reduce .[] as $x ([]; if any(.[]; . == $x) then . else . + [$x] end);
    def merge(a; b):
      if (a | type) == "object" and (b | type) == "object" then
        reduce (b | keys_unsorted[]) as $k (a; .[$k] = merge(a[$k]; b[$k]))
      elif (a | type) == "array" and (b | type) == "array" then (a + b | dedupe)
      else b end;
    merge(.[0]; .[1])
    | del(.description)
    | .name = $name
    | if .build then del(.image) else . end
    | .mounts = ((.mounts // []) | map(if type == "string" then gsub("__VSCWS_CLAUDE_DIR__"; $claude_dir) else . end))
    | if $mode == "remote" then .runArgs = (((.runArgs // []) + ["--network=host"]) | dedupe) else . end
  ' "$PRESETS_DIR/_common.json" "$pdir/devcontainer.json" > "$out/devcontainer.json"
  # Copy everything else the preset ships (Dockerfile, scripts).
  for f in "$pdir"/*; do
    [ -e "$f" ] || continue
    [ "$(basename "$f")" = devcontainer.json ] && continue
    cp -R "$f" "$out/"
  done
}

cmd_new() {
  need jq
  local name="" preset="" repo="" ws tmp answer
  while [ $# -gt 0 ]; do
    case "$1" in
      --preset) [ $# -ge 2 ] || die "--preset needs a value"; preset="$2"; shift 2 ;;
      --repo)   [ $# -ge 2 ] || die "--repo needs a value";   repo="$2";   shift 2 ;;
      -*)       die "unknown option '$1' (usage: vscws new NAME --preset P [--repo URL])" ;;
      *)        [ -z "$name" ] || die "unexpected argument '$1'"; name="$1"; shift ;;
    esac
  done
  [ -n "$name" ]   || die "usage: vscws new NAME --preset P [--repo URL]"
  validate_name "$name"
  [ -n "$preset" ] || die "--preset is required (see: vscws presets)"
  [ -f "$(preset_dir "$preset")/devcontainer.json" ] || die "unknown preset '$preset' (see: vscws presets)"
  ensure_root
  ws="$(ws_path "$name")"
  [ ! -e "$ws" ] || die "workspace already exists: $ws"
  mkdir -p "$VSCWS_ROOT"
  tmp="$(mktemp -d "$VSCWS_ROOT/.vscws-tmp.XXXXXX")"
  # shellcheck disable=SC2064
  trap "rm -rf '$tmp'" EXIT
  if [ -n "$repo" ]; then
    die "--repo is not implemented yet"
  else
    mkdir "$tmp/ws"
  fi
  generate_devcontainer "$preset" "$name" "$tmp/ws/.devcontainer"
  jq -n --arg name "$name" --arg preset "$preset" \
        --arg created "$(date -u +%Y-%m-%dT%H:%M:%SZ)" --arg v "$VSCWS_VERSION" \
```

```bash
cmd_ls() {
  need jq
  ensure_root
  local d name preset found=0
  if [ -d "$VSCWS_ROOT" ]; then
    for d in "$VSCWS_ROOT"/*/; do
      [ -d "$d" ] || continue
      name="$(basename "$d")"
      if [ -f "$d/.vscws.json" ]; then preset="$(jq -r '.preset // "?"' "$d/.vscws.json")"; else preset="-"; fi
      printf '%-24s %s\n' "$name" "$preset"
      found=1
    done
  fi
  [ "$found" -eq 1 ] || info "no workspaces in $VSCWS_ROOT"
}

cmd_open() {
  local name="${1:-}" ws
  [ -n "$name" ] || die "usage: vscws open NAME"
  validate_name "$name"
  ensure_root
  ws="$(ws_path "$name")"
  [ -d "$ws" ] || die "no such workspace: $ws"
  if [ "$VSCWS_MODE" = remote ]; then
    [ -n "$VSCWS_SSH_HOST" ] || die "VSCWS_SSH_HOST is not set in $CONFIG_FILE (the Host alias from your laptop's ~/.ssh/config)"
    info "Run this on your laptop, then choose \"Reopen in Container\" when VS Code asks:"
    info "  code --folder-uri \"vscode-remote://ssh-remote+$VSCWS_SSH_HOST$ws\""
  elif command -v code >/dev/null 2>&1; then
    code "$ws"
    info "opened $ws in VS Code; choose \"Reopen in Container\" when asked"
  else
    info "Open this folder in VS Code, then choose \"Reopen in Container\":"
    info "  $ws"
  fi
}

cmd_rm() {
  local name="${1:-}" ws typed ids
  [ -n "$name" ] || die "usage: vscws rm NAME"
  validate_name "$name"
  ensure_root
  ws="$(ws_path "$name")"
  [ -d "$ws" ] || die "no such workspace: $ws"
  printf 'This deletes %s and its dev container. Type the workspace name to confirm: ' "$ws"
  read -r typed || typed=""
  [ "$typed" = "$name" ] || die "aborted, nothing deleted"
  if command -v docker >/dev/null 2>&1; then
    ids="$(docker ps -aq --filter "label=devcontainer.local_folder=$ws" 2>/dev/null || true)"
    if [ -n "$ids" ]; then
      # shellcheck disable=SC2086
      docker rm -f $ids >/dev/null && info "removed dev container"
    fi
  fi
  rm -rf "$ws"
  info "deleted $ws"
}
```

````markdown
# vscws

Creates a per-project VS Code dev container workspace from a language preset. One command, then "Reopen in Container".

- Each workspace is a directory under a root you choose. It is mounted into the container at the same path.
- The container has the language's tools, all at the latest stable version at build time.
- VS Code installs only the extensions listed in the preset.
- The Claude Code panel works in every container with one shared login.
- Your project's own `docker compose` runs from inside the container against the host Docker.

## 1. Prerequisites

### On the machine that runs Docker (Linux VM)

- Docker (official installer, any recent Ubuntu):

```bash
curl -fsSL https://get.docker.com | sudo sh
sudo usermod -aG docker "$USER"
# log out and back in, then:
docker run --rm hello-world
```

- git and jq:

```bash
sudo apt-get update && sudo apt-get install -y git jq
```

- Claude Code (so the login can be shared into containers):

```bash
curl -fsSL https://claude.ai/install.sh | bash
claude   # log in once
```

### On a Mac that runs Docker locally instead

```bash
brew install git jq
brew install --cask docker     # then launch Docker Desktop once
curl -fsSL https://claude.ai/install.sh | bash
```

### On the laptop with VS Code

- Install the Remote-SSH and Dev Containers extensions:

```bash
code --install-extension ms-vscode-remote.remote-ssh
code --install-extension ms-vscode-remote.remote-containers
```

- Add the contents of `mac/settings.json` to your VS Code user settings (Cmd+Shift+P → "Preferences: Open User Settings (JSON)"). This keeps laptop extensions out of the containers.
- Add the VM to `~/.ssh/config`:

```
Host myvm
    HostName <vm ip>
    User <vm user>
    IdentityFile ~/.ssh/id_ed25519
    ForwardAgent yes
```

## 2. Install vscws

```bash
git clone <this repo url> ~/vscws
mkdir -p ~/.local/bin && ln -sf ~/vscws/bin/vscws ~/.local/bin/vscws
# make sure ~/.local/bin is on PATH, then:
vscws --version
```

- Config file `~/.config/vscws/config` (created on first `vscws new`, or write it yourself):

```bash
VSCWS_ROOT='/data/ws'        # where workspaces live
VSCWS_SSH_HOST='myvm'        # Host alias from the laptop's ~/.ssh/config (Linux VM only)
VSCWS_CLAUDE_DIR="$HOME/.claude"   # Claude config dir shared into containers
```

## 3. Share the Claude login with containers (Linux VM, optional)

- Claude keeps its login in two places: `~/.claude/` and the file `~/.claude.json`. Containers mount `~/.claude/`, so the file must live inside it.
- Run once (backs up first, leaves a symlink so nothing else breaks):

```bash
mkdir -p ~/.claude/backups && cp ~/.claude.json ~/.claude/backups/claude.json.$(date +%s)
mv ~/.claude.json ~/.claude/.claude.json && ln -s ~/.claude/.claude.json ~/.claude.json
printf '\nexport CLAUDE_CONFIG_DIR="$HOME/.claude"\n' >> ~/.profile
sed -i '1i export CLAUDE_CONFIG_DIR="$HOME/.claude"' ~/.bashrc
```

- Open a new shell and run `claude` to confirm you are still logged in.
- Skip this step if you prefer: the first container will ask you to log in once, and all containers share that login.
- On a Mac, skip it: the Mac keeps the login in Keychain. The first container asks for one login, shared by all containers after that.

## 4. Create and open a workspace

```bash
vscws presets                              # see what is available
vscws new myapi --preset go                # empty project
vscws new myapi --preset go --repo git@github.com:you/myapi.git   # or clone one
vscws open myapi                           # prints the command to run on the laptop
```

- On the laptop, run the printed `code --folder-uri ...` command. VS Code connects over SSH and asks "Reopen in Container". Click it.
- First open builds the image. Later opens take seconds.
- The Claude panel appears in the sidebar. The `claude` command also works in the VS Code terminal.
- On a Mac with local Docker, `vscws open` opens the folder directly.

## 5. Run your project's own Docker containers

- Open a terminal in VS Code (inside the dev container) and run `docker compose up` as usual. It runs on the host Docker.
- Relative bind mounts in compose files work because the workspace path is the same inside and outside.
- Linux VM: services published on a port are reachable at `localhost:<port>` from the dev container.
- Mac: use `host.docker.internal:<port>` instead of `localhost`.

## 6. Update a language configuration

- Edit `presets/<lang>/devcontainer.json` (features, extensions, settings) or `presets/<lang>/Dockerfile`.
- New workspaces pick it up immediately.
- Existing workspace: copy the changed file into its `.devcontainer/`, or edit `.devcontainer/devcontainer.json` there directly. Then in VS Code: Cmd+Shift+P → "Dev Containers: Rebuild Container".

## 7. Update tool and language versions

- Nothing is pinned. To move a workspace to the current latest stable of everything: Cmd+Shift+P → "Dev Containers: Rebuild Container Without Cache".
````

```bash
#!/usr/bin/env bash
# Manual integration test: builds a workspace for PRESET with the devcontainer CLI and checks tools inside.
# Usage: tests/integration.sh PRESET   (needs docker, devcontainer CLI, and VSCWS_ROOT configured)
set -euo pipefail
preset="${1:?usage: integration.sh PRESET}"
HERE="$(cd "$(dirname "$0")" && pwd -P)"
VSCWS="$HERE/../bin/vscws"
name="it-$preset"
root="${VSCWS_ROOT:-$(. "$HOME/.config/vscws/config"; echo "$VSCWS_ROOT")}"
ws="$root/$name"

if [ -d "$ws" ]; then printf '%s\n' "$name" | "$VSCWS" rm "$name"; fi
"$VSCWS" new "$name" --preset "$preset"

devcontainer up --workspace-folder "$ws"
x() { devcontainer exec --workspace-folder "$ws" bash -lc "$*"; }

echo "== common"
x 'whoami; pwd; echo CLAUDE_CONFIG_DIR=$CLAUDE_CONFIG_DIR'
x 'test "$(pwd)" = "'"$ws"'"'
x 'docker ps >/dev/null && echo docker-socket-ok'
x 'node --version; gh --version | head -1; claude --version'
x 'claude -p "reply with the single word ok" --max-turns 1'
x 'ls ~/.claude/.claude.json ~/.claude/.credentials.json'

echo "== $preset"
case "$preset" in
  go)     x 'go version; gopls version; dlv version | head -1; golangci-lint version' ;;
  java)   x 'java -version; mvn -v | head -1; gradle -v | grep Gradle' ;;
  cpp)    x 'gcc --version | head -1; clang --version | head -1; clangd --version; lldb --version; cmake --version | head -1; ninja --version; ccache --version | head -1' ;;

  bun)    x 'bun --version; node --version' ;;
  python) x 'python3 --version; uv --version; ruff --version' ;;
  *) echo "no checks for $preset"; exit 1 ;;
esac
echo "== $preset OK"
```
